cmake_minimum_required(VERSION 3.13)
set(PICO_BOARD a8_gb_board)
list(APPEND PICO_BOARD_HEADER_DIRS "${CMAKE_CURRENT_LIST_DIR}/../../boards")
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(dual_selector C CXX ASM)
pico_sdk_init()
add_executable(dual_selector selector.c entry.S)
target_include_directories(dual_selector PRIVATE .)
target_link_libraries(dual_selector pico_stdlib hardware_clocks hardware_sync hardware_watchdog)
pico_enable_stdio_uart(dual_selector 0)
pico_enable_stdio_usb(dual_selector 0)
pico_set_linker_script(dual_selector ${CMAKE_CURRENT_LIST_DIR}/memmap_slot.ld)
target_compile_options(dual_selector PRIVATE -O2 -Wall -Wextra -fstack-usage)
pico_add_extra_outputs(dual_selector)
set_target_properties(dual_selector PROPERTIES LINKER_LANGUAGE C)
target_link_options(dual_selector PRIVATE -Wl,--undefined=selector_entry)
